from pathlib import Path
import hashlib
import json
import sys

root = Path(r'D:\Gas-Sensor-Drift-Classification')
sys.path.insert(0, str(root))
from src.webapp.service import analyze, read_upload, load_models, downloads
import numpy as np
import pandas as pd

out = root / 'outputs/streamlit_ux'
models = load_models()
baseline = {}
for n in [99, 100, 101, 200]:
    for labels in [False, True]:
        frame = pd.DataFrame(np.random.default_rng(42).normal(size=(n,128)), columns=[f'f{i:03}' for i in range(1,129)])
        if labels:
            frame['gas'] = np.arange(n) % 6 + 1
        result = analyze(read_upload(frame.to_csv(index=False).encode('utf-8-sig')), models)
        baseline[f'{n}_{labels}'] = {name: hashlib.sha256(data).hexdigest() for name,data in downloads(result).items()}
(out / 'baseline_download_sha256.json').write_text(json.dumps(baseline, indent=2), encoding='utf-8')
(root/'src/webapp/ui.py').write_text(Path('stage195/ui.py').read_text(encoding='utf-8'),encoding='utf-8')
